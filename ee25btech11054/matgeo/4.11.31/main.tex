\begin{problem}[4.11.31]
Find the area of the region bounded by the line $y = 3x + 2$, the X axis and the ordinates $x = -2$ and $x = 1$.
\end{problem}

\solution
The total area is the sum of the areas of two triangles. To find their vertices, we must find the intersection points of the boundary lines: $y = 3x + 2$ (the line), $y = 0$ (the axis), $x = -2$, and $x = 1$.

\noindent\textbf{Finding the Vertices via Gaussian Elimination}

\noindent 1. \textbf{Intersection with the x-axis (Vertex B):}\\
This is the intersection of $3x - y = -2$ and $y=0$. The augmented matrix is:
\begin{equation}
    \left(\begin{array}{cc|c} 3 & -1 & -2 \\ 0 & 1 & 0 \end{array}\right)
    \xrightarrow{R_1 \to R_1 + R_2}
    \left(\begin{array}{cc|c} 3 & 0 & -2 \\ 0 & 1 & 0 \end{array}\right)
    \xrightarrow{R_1 \to R_1/3}
    \left(\begin{array}{cc|c} 1 & 0 & -2/3 \\ 0 & 1 & 0 \end{array}\right) \implies \vec{B} = \myvec{-2/3 \\ 0}
\end{equation}
\noindent 2. \textbf{Intersection with the ordinate $x=-2$ (Vertex D):}\\
This is the intersection of $3x - y = -2$ and $x=-2$. The augmented matrix is:
\begin{equation}
    \left(\begin{array}{cc|c} 3 & -1 & -2 \\ 1 & 0 & -2 \end{array}\right)
    \xrightarrow{R_1 \leftrightarrow R_2}
    \left(\begin{array}{cc|c} 1 & 0 & -2 \\ 3 & -1 & -2 \end{array}\right)
    \xrightarrow{R_2 \to R_2 - 3R_1}
    \left(\begin{array}{cc|c} 1 & 0 & -2 \\ 0 & -1 & 4 \end{array}\right)
    \xrightarrow{R_2 \to -R_2}
    \left(\begin{array}{cc|c} 1 & 0 & -2 \\ 0 & 1 & -4 \end{array}\right) \implies \vec{D} = \myvec{-2 \\ -4}
\end{equation}
\noindent 3. \textbf{Intersection with the ordinate $x=1$ (Vertex E):}\\
This is the intersection of $3x - y = -2$ and $x=1$. The augmented matrix is:
\begin{equation}
    \left(\begin{array}{cc|c} 3 & -1 & -2 \\ 1 & 0 & 1 \end{array}\right)
    \xrightarrow{R_1 \leftrightarrow R_2}
    \left(\begin{array}{cc|c} 1 & 0 & 1 \\ 3 & -1 & -2 \end{array}\right)
    \xrightarrow{R_2 \to R_2 - 3R_1}
    \left(\begin{array}{cc|c} 1 & 0 & 1 \\ 0 & -1 & -5 \end{array}\right)
    \xrightarrow{R_2 \to -R_2}
    \left(\begin{array}{cc|c} 1 & 0 & 1 \\ 0 & 1 & 5 \end{array}\right) \implies \vec{E} = \myvec{1 \\ 5}
\end{equation}
The remaining vertices are trivial intersections with the axes: $\vec{A} = \myvec{-2\\0}$ and $\vec{C} = \myvec{1\\0}$. These points form two triangles: $\triangle ABD$ and $\triangle EBC$.

\noindent\textbf{Area Calculation}\\
The total area is the sum of the areas of $\text{ar}(\triangle ABD)$ and $\text{ar}(\triangle EBC)$.
\begin{align}
    \text{ar}(\triangle ABD) &= \frac{1}{2}||\myvec{\vec{A}-\vec{B}} \times \myvec{\vec{A}-\vec{D}}|| = \frac{1}{2}||\myvec{-4/3 \\ 0} \times \myvec{0 \\ 4}|| = \frac{8}{3} \\
    \text{ar}(\triangle EBC) &= \frac{1}{2}||\myvec{\vec{E}-\vec{C}} \times \myvec{\vec{B}-\vec{C}}|| = \frac{1}{2}||\myvec{0 \\ 5} \times \myvec{-5/3 \\ 0}|| = \frac{25}{6}
\end{align}
The total area of the region is the sum of the two triangle areas.
\begin{equation}
    \text{Area of the region} = \frac{8}{3} + \frac{25}{6} = \frac{41}{6}
\end{equation}
