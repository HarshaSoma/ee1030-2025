\section{Problem}
\begin{frame}
\frametitle{Problem Statement}
Find the area of the region bounded by the line $y = 3x + 2$, the X axis and the ordinates $x = -2$ and $x = 1$.
\end{frame}

\section{Solution}
\subsection{Strategy}
\begin{frame}{Strategy}
The total area is the sum of the areas of two triangles. 
\begin{itemize}
    \item We first find the intersection points (vertices) of the boundary lines:
    \begin{itemize}
        \item $y = 3x + 2$
        \item $y = 0$ (the x-axis)
        \item $x = -2$
        \item $x = 1$
    \end{itemize}
    \item Then, we calculate the area of the two triangles formed by these vertices.
\end{itemize}
\end{frame}

\subsection{Finding Vertices}
\begin{frame}{Finding Vertex B (Intersection with x-axis)}
Intersection of $3x - y = -2$ and $y=0$. The augmented matrix is:
\begin{align*}
    \left(\begin{array}{cc|c} 3 & -1 & -2 \\ 0 & 1 & 0 \end{array}\right)
    &\xrightarrow{R_1 \to R_1 + R_2}
    \left(\begin{array}{cc|c} 3 & 0 & -2 \\ 0 & 1 & 0 \end{array}\right) \\
    &\xrightarrow{R_1 \to R_1/3}
    \left(\begin{array}{cc|c} 1 & 0 & -2/3 \\ 0 & 1 & 0 \end{array}\right)
\end{align*}
This gives the vertex $\vec{B} = \myvec{-2/3 \\ 0}$.
\end{frame}

\begin{frame}{Finding Vertex D (Intersection with $x=-2$)}
Intersection of $3x - y = -2$ and $x=-2$. The augmented matrix is:
\begin{align*}
    \left(\begin{array}{cc|c} 3 & -1 & -2 \\ 1 & 0 & -2 \end{array}\right)
    &\xrightarrow{R_1 \leftrightarrow R_2}
    \left(\begin{array}{cc|c} 1 & 0 & -2 \\ 3 & -1 & -2 \end{array}\right) \\
    &\xrightarrow{R_2 \to R_2 - 3R_1}
    \left(\begin{array}{cc|c} 1 & 0 & -2 \\ 0 & -1 & 4 \end{array}\right) \\
    &\xrightarrow{R_2 \to -R_2}
    \left(\begin{array}{cc|c} 1 & 0 & -2 \\ 0 & 1 & -4 \end{array}\right)
\end{align*}
This gives the vertex $\vec{D} = \myvec{-2 \\ -4}$.
\end{frame}

\begin{frame}{Finding Vertex E (Intersection with $x=1$)}
Intersection of $3x - y = -2$ and $x=1$. The augmented matrix is:
\begin{align*}
 \left(\begin{array}{cc|c} 3 & -1 & -2 \\ 1 & 0 & 1 \end{array}\right)
    &\xrightarrow{R_1 \leftrightarrow R_2}
    \left(\begin{array}{cc|c} 1 & 0 & 1 \\ 3 & -1 & -2 \end{array}\right) \\
    &\xrightarrow{R_2 \to R_2 - 3R_1}
    \left(\begin{array}{cc|c} 1 & 0 & 1 \\ 0 & -1 & -5 \end{array}\right) \\
    &\xrightarrow{R_2 \to -R_2}
    \left(\begin{array}{cc|c} 1 & 0 & 1 \\ 0 & 1 & 5 \end{array}\right)
\end{align*}
This gives the vertex $\vec{E} = \myvec{1 \\ 5}$.
\end{frame}

\begin{frame}{Summary of Vertices}
The remaining vertices are the intersections with the x-axis:
\begin{itemize}
    \item $\vec{A} = \myvec{-2\\0}$
    \item $\vec{C} = \myvec{1\\0}$
\end{itemize}
\vspace{1cm}
These points form two triangles: $\triangle ABD$ and $\triangle EBC$.
\end{frame}

\subsection{Area Calculation}
\begin{frame}{Area Calculation}
The total area is the sum of the areas of $\triangle ABD$ and $\triangle EBC$.
\vspace{1cm}
Area of $\triangle ABD$:
\begin{align}
    \text{ar}(\triangle ABD) &= \frac{1}{2}\left|\left|\myvec{\vec{A}-\vec{B}} \times \myvec{\vec{A}-\vec{D}}\right|\right| \\
    &= \frac{1}{2}\left|\left|\myvec{-4/3 \\ 0} \times \myvec{0 \\ 4}\right|\right| = \frac{8}{3}
\end{align}
\vspace{1cm}
Area of $\triangle EBC$:
\begin{align}
    \text{ar}(\triangle EBC) &= \frac{1}{2}\left|\left|\myvec{\vec{E}-\vec{C}} \times \myvec{\vec{B}-\vec{C}}\right|\right| \\
    &= \frac{1}{2}\left|\left|\myvec{0 \\ 5} \times \myvec{-5/3 \\ 0}\right|\right| = \frac{25}{6}
\end{align}
\end{frame}

\subsection{Total Area}
\begin{frame}{Total Area}
The total area of the region is the sum of the two triangle areas.
\begin{equation}
    \text{Area} = \text{ar}(\triangle ABD) + \text{ar}(\triangle EBC)
\end{equation}
\begin{equation}
    \text{Area} = \frac{8}{3} + \frac{25}{6} = \frac{16+25}{6} = \frac{41}{6}
\end{equation}
\end{frame}
