\section{Problem}
\begin{frame}
\frametitle{Problem Statement}
Find the coordinates of the point where the line through \brak{5, 1, 6} and \brak{3, 4, 1} crosses the $YZ$-plane.
\end{frame}

\section{Solution}
\begin{frame}{Given data}
The two points on the line are given by the position vectors:
\begin{align}
 \vec{A} = \myvec{5\\1\\6} \\
 \vec{B} = \myvec{3\\4\\1}
\end{align}
\end{frame}

\begin{frame}{Formulae}
The vector equation of a line passing through points $\vec{A}$ and $\vec{B}$ is given by:
\begin{align}
 \vec{x} &= \vec{A} + k\brak{\vec{B}-\vec{A}}
\end{align}
where $k$ is a scalar parameter.

The equation of the $YZ$-plane is $x=0$. A point $\vec{x}$ lies on this plane if its $x$-component is zero. This can be expressed as:
\begin{align}
 \vec{e_1}^\top\vec{x} &= 0
\end{align}
where $\vec{e_1} = \myvec{1\\0\\0}$.
\end{frame}

\begin{frame}{Solving}
First, find the direction vector of the line:
\begin{align}
 \vec{B} - \vec{A} = \myvec{3-5 \\ 4-1 \\ 1-6} = \myvec{-2 \\ 3 \\ -5}
\end{align}
The parametric equation of the line is:
\begin{align}
 \vec{x} = \myvec{5 \\ 1 \\ 6} + k \myvec{-2 \\ 3 \\ -5} = \myvec{5-2k \\ 1+3k \\ 6-5k}
\end{align}
To find the intersection with the $YZ$-plane, we set the $x$-component to zero:
\begin{align}
  5-2k = 0 \implies 2k=5 \implies k = \frac{5}{2}
\end{align}
Substitute $k = 5/2$ back into the line equation to find the point:
\begin{align}
 \vec{x} = \myvec{5-2(5/2) \\ 1+3(5/2) \\ 6-5(5/2)} = \myvec{0 \\ 1+15/2 \\ 6-25/2} = \myvec{0 \\ 17/2 \\ -13/2}
\end{align}
\end{frame}
