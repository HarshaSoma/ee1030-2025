\begin{frame}{Problem Statement}
Find the area of the circle $x^2 + y^2 = 16$ exterior to the parabola $y^2 = 6x$.
\end{frame}

%---------------- Solution ----------------
\begin{frame}{Solution: Conic Parameters}
The total area of the circle $x^2 + y^2 = 16$ is $A_{circle} = \pi r^2 = 16\pi$.
First, we find the points of intersection. The conic parameters are:
\pause
For the circle $x^2 + y^2 - 16 = 0$:
\begin{align}
\mtx{V_1} = \myvec{1 & 0 \\ 0 & 1}, \quad \vec{u_1} = \myvec{0 \\ 0}, \quad f_1 = -16. \tag{1}
\end{align}
\pause
For the parabola $y^2 - 6x = 0$:
\begin{align}
\mtx{V_2} = \myvec{0 & 0 \\ 0 & 1}, \quad \vec{u_2} = \myvec{-3 \\ 0}, \quad f_2 = 0. \tag{2}
\end{align}
\end{frame}

\begin{frame}{Solution: Intersection of Conics}
The intersection of the two conics is given by the equation:
\begin{equation}
	\vec{x}^\top(\mtx{V_1} + \mu\mtx{V_2})\vec{x} + 2(\vec{u_1} + \mu\vec{u_2})^\top\vec{x} + (f_1 + \mu f_2) = 0 \tag{3}
\end{equation}
\pause
For a degenerate conic, the determinant of the quadratic part's matrix must be zero.
\begin{align}
\det(\mtx{V_1} + \mu\mtx{V_2}) &= 0 \tag{4}
\end{align}
\end{frame}

\begin{frame}{Solution: Finding Parameter $\mu$}
\begin{align}
\det\left(\myvec{1 & 0 \\ 0 & 1} + \mu\myvec{0 & 0 \\ 0 & 1}\right) &=
\begin{vmatrix}
	1 & 0 \\
	0 & 1+\mu
\end{vmatrix}
 = 1(1+\mu) = 0 \tag{5} \\
 \pause
\implies \mu &= -1 \tag{6}
\end{align}
\end{frame}

\begin{frame}{Solution: Degenerate Conic Equation}
Substituting $\mu = -1$ into equation (3):
\begin{align} 
\vec{x}^\top( \mtx{V_1} - \mtx{V_2} )\vec{x} + 2( \vec{u_1} - \vec{u_2} )^\top\vec{x} + ( f_1 - f_2 ) &= 0 \tag{7} \\
\vec{x}^\top\left( \myvec{1 & 0 \\ 0 & 1} - \myvec{0 & 0 \\ 0 & 1} \right)\vec{x} + 2\left( \myvec{0 \\ 0} - \myvec{-3 \\ 0} \right)^\top\vec{x} + ( -16 - 0 ) &= 0 \tag{8} \\
\vec{x}^\top \myvec{1 & 0 \\ 0 & 0} \vec{x} + 2 \myvec{3 \\ 0}^\top \vec{x} - 16 &= 0 \tag{9}
\end{align}
\end{frame}

\begin{frame}{Solution: Solving for Intersection Points}
Letting $\vec{x} = \myvec{x \\ y}$, the equation becomes:
\begin{align}
\myvec{x & y} \myvec{1 & 0 \\ 0 & 0} \myvec{x \\ y} + 2 \myvec{3 & 0} \myvec{x \\ y} - 16 = 0 \tag{10} \\
\implies x^2 + 6x - 16 = 0 \tag{11}
\end{align}
Solving for $x$: $(x+8)(x-2) = 0$, which gives $x=2$ or $x=-8$.
\end{frame}

\begin{frame}{Solution: Intersection Points}
Substituting these values into $y^2 = 6x$:
\begin{itemize}
    \item If $x=2$, $y^2 = 12 \implies y = \pm\sqrt{12} = \pm 2\sqrt{3}$.
    \item If $x=-8$, $y^2 = -48$, which has no real solution for $y$.
\end{itemize}
\pause
Hence, the points of intersection are:
\begin{align} 
\vec{a_1} = \myvec{2 \\ 2\sqrt{3}}, \quad \vec{a_2} = \myvec{2 \\ -2\sqrt{3}} \tag{12}
\end{align}
\end{frame}

\begin{frame}{Solution: Area Calculation Setup}
The area of the circle interior to the parabola is found by integration:
\[ A_{interior} = 2 \left[ \int_{0}^{2} \sqrt{6x} \,dx + \int_{2}^{4} \sqrt{16 - x^2} \,dx \right] \tag{13} \]
\pause
First integral:
\begin{align}
\int_{0}^{2} \sqrt{6x} \,dx &= \sqrt{6} \left[ \frac{2}{3}x^{3/2} \right]_0^2 = \frac{2\sqrt{6}}{3}(2\sqrt{2}) \nonumber \\
&= \frac{4\sqrt{12}}{3} = \frac{8\sqrt{3}}{3} \tag{14}
\end{align}
\end{frame}

\begin{frame}{Solution: Area Calculation}
Second integral (using trig substitution):
\begin{align}
&\int_{2}^{4} \sqrt{16-x^2} \,dx = \left[ \frac{x}{2}\sqrt{16-x^2} + 8\sin^{-1}\left(\frac{x}{4}\right) \right]_2^4 \nonumber \\
&= \left( 0 + 8\sin^{-1}(1) \right) - \left( \sqrt{12} + 8\sin^{-1}\left(\frac{1}{2}\right) \right) \nonumber \\
&= 8\left(\frac{\pi}{2}\right) - \left(2\sqrt{3} + 8\left(\frac{\pi}{6}\right)\right) = 4\pi - 2\sqrt{3} - \frac{4\pi}{3} \nonumber \\
&= \frac{8\pi}{3} - 2\sqrt{3} \tag{15}
\end{align}
\end{frame}

\begin{frame}{Solution: Final Area}
Adding the results for $A_{interior}$:
\begin{align}
A_{interior} &= 2 \left[ \frac{8\sqrt{3}}{3} + \frac{8\pi}{3} - 2\sqrt{3} \right] \nonumber \\ 
&= 2 \left[ \frac{2\sqrt{3}}{3} + \frac{8\pi}{3} \right] = \frac{4\sqrt{3}}{3} + \frac{16\pi}{3} \tag{16}
\end{align}
\pause
The required area exterior to the parabola is $A_{exterior} = A_{circle} - A_{interior}$.
\begin{align}
A_{exterior} &= 16\pi - \left( \frac{4\sqrt{3}}{3} + \frac{16\pi}{3} \right) \nonumber \\
&= \frac{48\pi - 16\pi}{3} - \frac{4\sqrt{3}}{3} = \frac{32\pi - 4\sqrt{3}}{3} \tag{17}
\end{align}
\end{frame}

\begin{lstlisting}[language=C]
#include <math.h>

double integrand(double y) {
    if (16.0 - y * y < 0) {
        return 0.0;
    }
    return sqrt(16.0 - y * y) - (y * y / 6.0);
}

double calculate_common_area(double a, double b, int n) {
    double h = (b - a) / n;
    double sum = integrand(a) + integrand(b);

    for (int i = 1; i < n; i += 2) {
        sum += 4 * integrand(a + i * h);
    }

    for (int i = 2; i < n; i += 2) {
        sum += 2 * integrand(a + i * h);
    }

    return (h / 3.0) * sum;
}
\end{lstlisting}
