%---------------- Problem Statement ----------------
\begin{frame}{Problem Statement}
A father's age is three times the sum of the ages of his two children. After 5 years his age will be two times the sum of their ages. Find the present age of the father.
\end{frame}

%---------------- Solution ----------------
\begin{frame}{Solution: Setting up Equations}
Let the present age of the father be $F$.
Let the sum of the present ages of his two children be $S$.
\begin{align}
F &= 3S \tag{1} \\
F - 3S &= 0 \tag{2}
\end{align}
\end{frame}

\begin{frame}{Solution: Equations after 5 years}
After 5 years:
\begin{align}
F + 5 &= 2(S + 10) \tag{3} \\
F + 5 &= 2S + 20 \tag{4} \\
F - 2S &= 15 \tag{5}
\end{align}
\end{frame}

\begin{frame}{Solution: Matrix Form}
We now have a system of two linear equations. In matrix form $\mtx{A}\vec{x} = \vec{b}$, this is:
\begin{align}
\myvec{1 & -3 \\ 1 & -2} \myvec{F \\ S} &= \myvec{0 \\ 15} \tag{6}
\end{align}
To solve this system, we use Gaussian elimination on the augmented matrix $[\mtx{A}|\vec{b}]$.
\begin{align}
\augvec{2}{1}{1 & -3 & 0 \\ 1 & -2 & 15} \tag{7}
\end{align}
\end{frame}

\begin{frame}{Solution: Gaussian Elimination}
Perform the row operation $R_2 \to R_2 - R_1$:
\begin{align}
\augvec{2}{1}{1 & -3 & 0 \\ 0 & 1 & 15} \tag{8}
\end{align}
From the second row, we get:
\begin{align}
S = 15 \tag{9}
\end{align}
Now, we perform back-substitution. Perform the row operation $R_1 \to R_1 + 3R_2$:
\begin{align}
\augvec{2}{1}{1 & 0 & 45 \\ 0 & 1 & 15} \tag{10}
\end{align}
\end{frame}

\begin{frame}{Solution: Final Answer}
From the reduced row echelon form, we get the solution:
\begin{align}
F = 45 \quad \text{and} \quad S = 15 \tag{11}
\end{align}
The question asks for the present age of the father.
\vspace{1em}
Thus, the present age of the father is \textbf{45 years}.
\end{frame}

\section{C Code}

#include <math.h>

int solve_system(double A[2][2], double b[2], double result[2]) {
    double det = A[0][0] * A[1][1] - A[0][1] * A[1][0];

    if (fabs(det) < 1e-9) {
        return 0;
    }

    double inv_det = 1.0 / det;
    double A_inv[2][2];
    A_inv[0][0] =  A[1][1] * inv_det;
    A_inv[0][1] = -A[0][1] * inv_det;
    A_inv[1][0] = -A[1][0] * inv_det;
    A_inv[1][1] =  A[0][0] * inv_det;

    result[0] = A_inv[0][0] * b[0] + A_inv[0][1] * b[1];
    result[1] = A_inv[1][0] * b[0] + A_inv[1][1] * b[1];
    
    return 1;
}
\end{lstlisting}
\end{frame}

b_c = (ctypes.c_double*2)(*b)

import numpy as np

A = np.array([[1, -3], [1, -2]])
b = np.array([0, 15])

try:
    solution = np.linalg.solve(A, b)
    fathers_age = solution[0]
    sum_of_childrens_ages = solution[1]
    
    print("--- Verification using NumPy ---")
    print(f"Father's present age: {fathers_age:.0f}")
    print(f"Sum of children's present ages: {sum_of_childrens_ages:.0f}")

    assert abs(fathers_age - 45) < 1e-9
    assert abs(sum_of_childrens_ages - 15) < 1e-9
    print("\nResult is correct.")

except np.linalg.LinAlgError:
    print("The matrix is singular. NumPy could not find a unique solution.")
