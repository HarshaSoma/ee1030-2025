\begin{frame}
\frametitle{Question}
Find a unit vector perpendicular to both the vectors $\vec{a}$ and $\vec{b}$, where:
\begin{align}
\vec{a} &= \hat{i} - 7\hat{j} + 7\hat{k} \label{eq:a_vec}\\
\vec{b} &= 3\hat{i} - 2\hat{j} + 2\hat{k} \label{eq:b_vec}
\end{align}
\end{frame}

\begin{frame}[t]{Solution}
\small
\textbf{Step 1: Setting up the equations}

Let the required vector be $\vec{n} = n_1\hat{i} + n_2\hat{j} + n_3\hat{k}$. For $\vec{n}$ to be perpendicular to both $\vec{a}$ and $\vec{b}$, their dot products must be zero.
\begin{align}
\vec{a} \cdot \vec{n} &= 0 \implies n_1 - 7n_2 + 7n_3 = 0 \label{eq:orth1}\\
\vec{b} \cdot \vec{n} &= 0 \implies 3n_1 - 2n_2 + 2n_3 = 0 \label{eq:orth2}
\end{align}

This forms a system of linear equations. In matrix form:
\begin{align}
\myvec{
1 & -7 & 7\\
3 & -2 & 2
}
\myvec{n_1\\n_2\\n_3}
= \myvec{0\\0} \label{eq:system}
\end{align}
\end{frame}

\begin{frame}
\small
\textbf{Step 2: Solving the system using row reduction}

We form the augmented matrix and perform row operations.
\begin{align}
&\myvec{
1 & -7 & 7 & 0\\
3 & -2 & 2 & 0
} \label{eq:aug} \\
\xrightarrow{R_2 \to R_2 - 3R_1} \quad &\myvec{
1 & -7 & 7 & 0\\
0 & 19 & -19 & 0
} \label{eq:row1} \\
\xrightarrow{R_2 \to \tfrac{1}{19}R_2} \quad &\myvec{
1 & -7 & 7 & 0\\
0 & 1 & -1 & 0
} \label{eq:row2} \\
\xrightarrow{R_1 \to R_1 + 7R_2} \quad &\myvec{
1 & 0 & 0 & 0\\
0 & 1 & -1 & 0
} \label{eq:row3}
\end{align}
\end{frame}

\begin{frame}
\small
\textbf{Step 3: Finding the general solution}

From the row-reduced echelon form \eqref{eq:row3}, we get the equations:
\begin{align}
n_1 &= 0 \label{eq:n1}\\
n_2 - n_3 &= 0 \implies n_2 = n_3 \label{eq:n2n3}
\end{align}
Let $n_3 = t$, where $t$ is a real number. Then $n_2 = t$. The general solution for a vector perpendicular to $\vec{a}$ and $\vec{b}$ is:
\begin{align}
\vec{n} = \myvec{0\\t\\t} = t\myvec{0\\1\\1}, \quad t\in\mathbb{R} \label{eq:gensol}
\end{align}
\end{frame}

\begin{frame}
\textbf{Step 4: Finding the unit vector}

To find the unit vector, we first find the magnitude of the direction vector $\myvec{0\\1\\1}$:
\begin{align}
\norm{\myvec{0\\1\\1}} = \sqrt{0^2 + 1^2 + 1^2} = \sqrt{2} \label{eq:norm}
\end{align}

Now, we normalize the vector $\vec{n}$ by dividing by its magnitude. The two possible unit vectors are:
\begin{align}
\hat{n} &= \pm \frac{1}{\sqrt{2}}\myvec{0\\1\\1} \label{eq:unit1}\\
&= \pm \frac{1}{\sqrt{2}}(\hat{j}+\hat{k}) \label{eq:unit2}
\end{align}
\end{frame}
