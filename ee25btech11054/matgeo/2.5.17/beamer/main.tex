\begin{frame}
\frametitle{Question}
Find a unit vector perpendicular to both the vectors $\vec{a}$ and $\vec{b}$, where
\begin{align}
\vec{a} &= \hat{i} - 7\hat{j} + 7\hat{k} \label{eq:a_vec}\\
\vec{b} &= 3\hat{i} - 2\hat{j} + 2\hat{k} \label{eq:b_vec}
\end{align}
\end{frame}

\begin{frame}[t]{Solution}
\small
\textbf{Step 1: Strategy}

A vector perpendicular to both $\vec{a}$ and $\vec{b}$ is given by their cross product, $\vec{c} = \vec{a} \times \vec{b}$.

To find the unit vector, we will first compute the cross product and then normalize it by dividing by its magnitude.
\begin{align}
    \hat{n} = \frac{\vec{a} \times \vec{b}}{\norm{\vec{a} \times \vec{b}}}
\end{align}

\textbf{Step 2: Computing the Cross Product}

The cross product can be calculated using the determinant of a matrix:
\begin{align}
\vec{a} \times \vec{b} &= \begin{vmatrix}
\hat{i} & \hat{j} & \hat{k} \\
1 & -7 & 7 \\
3 & -2 & 2
\end{vmatrix} \label{eq:cross_det}
\end{align}
\end{frame}

\begin{frame}
\small
Expanding the determinant from \eqref{eq:cross_det}:
\begin{align}
\vec{a} \times \vec{b} &= \hat{i}\begin{vmatrix} -7 & 7 \\ -2 & 2 \end{vmatrix} - \hat{j}\begin{vmatrix} 1 & 7 \\ 3 & 2 \end{vmatrix} + \hat{k}\begin{vmatrix} 1 & -7 \\ 3 & -2 \end{vmatrix} \nonumber \\
&= \hat{i}\brak{(-7)(2) - (7)(-2)} - \hat{j}\brak{(1)(2) - (7)(3)} + \hat{k}\brak{(1)(-2) - (-7)(3)} \nonumber \\
&= \hat{i}\brak{-14 + 14} - \hat{j}\brak{2 - 21} + \hat{k}\brak{-2 + 21} \nonumber \\
&= 0\hat{i} - \brak{-19}\hat{j} + 19\hat{k} \nonumber \\
&= 19\hat{j} + 19\hat{k} \label{eq:cross_result}
\end{align}
Let's call this resulting vector $\vec{c} = 19\hat{j} + 19\hat{k}$.
\end{frame}

\begin{frame}
\small
\textbf{Step 3: Finding the Magnitude}

Now, we compute the magnitude of the cross product vector $\vec{c}$:
\begin{align}
\norm{\vec{a} \times \vec{b}} &= \norm{\vec{c}} = \sqrt{(0)^2 + (19)^2 + (19)^2} \nonumber \\
&= \sqrt{2 \times 19^2} \nonumber \\
&= 19\sqrt{2} \label{eq:magnitude}
\end{align}

\textbf{Step 4: Normalizing to get the Unit Vector}

Finally, we find the unit vector $\hat{n}$ by dividing the vector $\vec{c}$ by its magnitude:
\begin{align}
\hat{n} &= \frac{\vec{a} \times \vec{b}}{\norm{\vec{a} \times \vec{b}}} = \frac{19\hat{j} + 19\hat{k}}{19\sqrt{2}} \nonumber \\
&= \frac{19(\hat{j} + \hat{k})}{19\sqrt{2}} \nonumber \\
&= \frac{1}{\sqrt{2}}\brak{\hat{j} + \hat{k}} \label{eq:unit_vector}
\end{align}
\end{frame}

\begin{frame}
\frametitle{Final Answer}
The unit vector perpendicular to both $\vec{a}$ and $\vec{b}$ is:
\begin{align}
\hat{n} &= \frac{1}{\sqrt{2}}\brak{\hat{j} + \hat{k}} \nonumber
\end{align}
or in component form:
\begin{align}
\hat{n} = \myvec{0 \\ \frac{1}{\sqrt{2}} \\ \frac{1}{\sqrt{2}}}
\end{align}
\end{frame}
