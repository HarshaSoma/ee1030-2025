\textbf{Question:}\par
$|\vec{a}| = 3, |\vec{b}| = 2\sqrt{3}, \text{ and } \vec{a}\cdot\vec{b} = 6$, then find the value of $|\vec{a}\times\vec{b}|$. \par
\textbf{Solution:}\par

Given:
\begin{align}
 \norm{\vec{a}} &= 3 \\
 \norm{\vec{b}} &= 2\sqrt{3} \\
 \vec{a}^\top\vec{b} &= 6
\end{align}
We know:
\begin{align}
 \norm{\vec{a}\times\vec{b}} &= \norm{\vec{a}}\norm{\vec{b}}\sin\theta \\
 \cos\theta &= \frac{\vec{a}^\top\vec{b}}{\norm{\vec{a}}\norm{\vec{b}}}
\end{align}
Thus
\begin{align}
 \brak{\vec{a}^\top\vec{b}}^2 + \brak{\norm{\vec{a}\times\vec{b}}}^2 = \norm{\vec{a}}^2\norm{\vec{b}}^2
\end{align}
Substituting values
\begin{align}
\norm{\vec{a}\times\vec{b}} &= \sqrt{(3^2)\times (2\sqrt{3})^2 - 6^2} \\
\norm{\vec{a}\times\vec{b}} &= \sqrt{108-36} \\
\norm{\vec{a}\times\vec{b}} &=  \sqrt{72} \\
\norm{\vec{a}\times\vec{b}} &=  6\sqrt{2} \\
\end{align}
